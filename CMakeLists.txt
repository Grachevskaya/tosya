cmake_minimum_required(VERSION 3.20)
project(tosya LANGUAGES NONE)

# Usage:
#   cmake -S . -B build -DDDK_ROOT=/opt/ddk
#   cmake --build build                     # default target: every KMI module
#   cmake --build build --target ko         # the same, by name
#   cmake --build build --target ko-android14-6.1
# Everything is written into build/ only: kmi/  ko/  dist/

set(DDK_ROOT "/opt/ddk" CACHE PATH "DDK root (provides kdir/ and clang/)")
option(TOSYA_USE_PREBUILT_KO "Package existing kernel modules from build/ko" OFF)
option(TOSYA_DEBUG "Build a module whose diagnostics are on from boot" OFF)

# KMI -> clang shipped with the DDK. Each KMI is built by the compiler its kernel was built with,
# which is also the compiler clang CFI and the shadow call stack were generated by.
set(TOSYA_KMI_CLANG
    android12-5.10:clang-r416183b
    android13-5.10:clang-r450784e
    android13-5.15:clang-r450784e
    android14-5.15:clang-r487747c
    android14-6.1:clang-r487747c
    android15-6.6:clang-r510928
    android16-6.12:clang-r536225
    android17-6.18:clang-r584948c)

set(KO_NAME tosya)

set(KO_DIR "${CMAKE_BINARY_DIR}/ko")
file(MAKE_DIRECTORY "${KO_DIR}")
file(GLOB SRC_FILES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.S"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/include/*.h")
list(APPEND SRC_FILES "${CMAKE_CURRENT_SOURCE_DIR}/src/Makefile")
list(APPEND SRC_FILES "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-kmi.sh")

# Kernel modules must be linked by kbuild (modpost, vermagic, symbol CRCs). kbuild
# writes its objects into the M= directory, so every KMI builds inside
# build/kmi/<kmi>/ -- a directory that only holds symlinks to src/. The source tree
# is therefore never written to, and no copy of it is needed either.
set(_ko_outputs "")
if(TOSYA_USE_PREBUILT_KO)
  file(GLOB _ko_outputs CONFIGURE_DEPENDS "${KO_DIR}/*.ko")
  if(NOT _ko_outputs)
    message(FATAL_ERROR "TOSYA_USE_PREBUILT_KO is enabled, but no kernel modules were found in ${KO_DIR}")
  endif()
else()
  # CMAKE_MAKE_PROGRAM can be Ninja. Kbuild always requires GNU make.
  find_program(TOSYA_KBUILD_MAKE NAMES gmake make REQUIRED)
  find_program(TOSYA_BASH NAMES bash REQUIRED)
  foreach(_pair IN LISTS TOSYA_KMI_CLANG)
    string(REPLACE ":" ";" _p "${_pair}")
    list(GET _p 0 _kmi)
    list(GET _p 1 _clang)
    set(_kdir "${DDK_ROOT}/kdir/${_kmi}")
    set(_clang_bin "${DDK_ROOT}/clang/${_clang}/bin")
    set(_stage "${CMAKE_BINARY_DIR}/kmi/${_kmi}")
    set(_out "${KO_DIR}/${_kmi}_arm64_${KO_NAME}.ko")

    if(NOT EXISTS "${_kdir}" OR NOT EXISTS "${_clang_bin}")
      message(FATAL_ERROR "missing DDK tree or clang for ${_kmi}: ${_kdir} / ${_clang_bin}")
    endif()

    add_custom_command(OUTPUT "${_out}"
      COMMAND ${CMAKE_COMMAND} -E env "DDK_ROOT=${DDK_ROOT}" "TOSYA_KBUILD_MAKE=${TOSYA_KBUILD_MAKE}"
              "${TOSYA_BASH}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-kmi.sh"
              "${_kmi}" "${_clang}" "${_stage}" "${_out}" "$<BOOL:${TOSYA_DEBUG}>"
      DEPENDS ${SRC_FILES}
      COMMENT "kbuild ${_kmi}"
      VERBATIM)
    add_custom_target("ko-${_kmi}" DEPENDS "${_out}")
    list(APPEND _ko_outputs "${_out}")
  endforeach()
endif()
add_custom_target(ko DEPENDS ${_ko_outputs})
